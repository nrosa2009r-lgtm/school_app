async def get_user_id_by_email(email: str) -> int | None:
    async with async_session() as session:
        # 1. Wybieramy tylko kolumnę id, a .where() dajemy POZA nawiasem tabeli
        query = select(Users.id).where(Users.email == email)
        
        # 2. Wykonujemy zapytanie
        result = await session.execute(query)
        
        # 3. Zwracamy pojedynczą wartość (id) lub None
        return result.scalar_one_or_none()
